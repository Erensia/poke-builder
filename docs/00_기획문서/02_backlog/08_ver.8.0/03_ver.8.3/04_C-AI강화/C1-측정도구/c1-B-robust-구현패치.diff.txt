+/**
+ * 후보 하나의 값(8.3 C1 Stage 1). `weights` = 응수 가중치, `sums` = 응수별 롤아웃 합, `seeds` = 시드 수. ρ ≤ 0이면 현행 기대값 식을 그대로 쓴다.

+export function robustValue(weights: number[], sums: number[], seeds: number, rho: number): number {
+  const expected = weights.reduce((v, weight, j) => v + (weight * sums[j]) / seeds, 0);
+  if (!(rho > 0)) return expected;
+  let worst = Infinity;
+  weights.forEach((weight, j) => {
+    if (weight >= ROBUST_MIN_WEIGHT) worst = Math.min(worst, sums[j] / seeds);
+  });
+  return Number.isFinite(worst) ? (1 - rho) * expected + rho * worst : expected;
 }
