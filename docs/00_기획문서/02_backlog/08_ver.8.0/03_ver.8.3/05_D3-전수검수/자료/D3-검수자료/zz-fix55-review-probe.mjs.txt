// fix-55 검수 엔진 프로브: 교체 → 스텔스록 기절 턴에 상대(b)가 각 기술을 쓸 때. 인자: poke-builder 경로
const { createServer } = await import(process.argv[2] + "/node_modules/vite/dist/node/index.js");
const root = process.argv[2];
const server = await createServer({ root, server: { middlewareMode: true, hmr: false }, appType: "custom", logLevel: "error" });
try {
  const data = await server.ssrLoadModule("/src/lib/data.ts");
  const S = await server.ssrLoadModule("/src/lib/battle/state.ts");
  const rt = await server.ssrLoadModule("/src/lib/battle/runTurn.ts");
  const pts = { hp: 2, atk: 32, def: 0, spa: 0, spd: 0, spe: 32 };
  const mon = (id, moves, ability = "두꺼운지방", item = null) => ({ slot: { pokemonId: id, moves: [...moves, null, null, null, null].slice(0, 4), ability, item, nature: null, points: pts }, moves: moves.map((m) => data.getMove(m)) });
  const battle = (A, B) => S.createBattleState({ a: { slots: A.map((m) => m.slot), movesList: A.map((m) => m.moves) }, b: { slots: B.map((m) => m.slot), movesList: B.map((m) => m.moves) } });
  const mv = (x) => ({ kind: "move", move: data.getMove(x) });
  const rnd = () => 0.5;
  for (const [label, species, move, item] of [["저주(고스트)", "팬텀", "저주"], ["록온", "잠만보", "록온"], ["문어굳히기", "잠만보", "문어굳히기"], ["미래예지", "잠만보", "미래예지"], ["구애스카프 아이언헤드", "잠만보", "아이언헤드", "구애스카프"], ["멸망의노래", "잠만보", "멸망의노래"]]) {
    try {
      let st = battle([mon("잠만보", ["잠꼬대"]), mon("잠만보", ["잠꼬대"]), mon("잠만보", ["잠꼬대"])], [mon(species, [move, "잠꼬대"], null, item ?? null)]);
      st.sideA.hazards = { ...st.sideA.hazards, stealthRock: true };
      st.sideA.party[1].currentHp = 1;
      let o = rt.runTurn(st, { kind: "switch", toIndex: 1 }, mv(move), rnd);
      const paused = "awaitingSelfSwitch" in o;
      for (let g = 0; "awaitingSelfSwitch" in o && g < 5; g++) o = rt.resumeTurn(o._ctx, 2);
      const n = o.nextState, b = n.b, act = o.result.actions.find((x) => x.actor === "b");
      console.log(`${label}: paused=${paused} forced=${JSON.stringify(o.forcedSwitch ?? null)} a활성=${n.sideA.activeIndex}(${n.a.currentHp}/${n.a.maxHp}) aVol=[${Object.keys(n.a.volatile.active)}]` +
        ` | b ${b.currentHp}/${b.maxHp} blocked=${act?.blockedReason ?? "-"} hit=${act?.hit} lastMove=${b.lastMoveId ?? "-"} choiceLock=${b.choiceLockedMoveId ?? "-"} bVol=[${Object.keys(b.volatile.active)}] bPerish=${b.perishCount ?? "-"} delayed=${n.sideA.futureSight ? n.sideA.futureSight.turnsRemaining : "-"}`);
    } catch (e) { console.log(label, "예외", e.message); }
  }
} finally { await server.close(); }
