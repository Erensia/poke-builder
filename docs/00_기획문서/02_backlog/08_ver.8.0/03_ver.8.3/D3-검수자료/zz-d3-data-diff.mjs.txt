  const o = []; let cur = "", q = false;
  for (const ch of l) { if (ch === '"') q = !q; else if (ch === "," && !q) { o.push(cur); cur = ""; } else cur += ch; }
  o.push(cur); return o;
});
const ns = (s) => s.replace(/\s+/g, "");
function nameMap(base, nameFile, idCol) {
  const ident = new Map(csv(base).map((r) => [r[0], r[1]]));
  const ko = new Map(); // 공백 제거 한국어 → showdown id 후보들
  for (const r of csv(nameFile)) if (r[1] === "3") {
    const sid = ident.get(r[0])?.replace(/-/g, ""); if (!sid) continue;
    const k = ns(r[2]); if (!ko.has(k)) ko.set(k, []); ko.get(k).push(sid);
  }
  return ko;
}

// ───── 매핑 ─────
function map(kind, engArr, ko, sdLegal, sdAll, manual = {}) {
  const res = new Map(); const unmatched = []; const multi = [];
  for (const e of engArr) {
    const k = ns(e.name);
    let c = manual[k] ? [manual[k]] : ko.get(k);
    if (!c) { unmatched.push(e.name); continue; }
    const legal = c.filter((x) => sdAll[x]);
    const pick = legal.length ? legal : c;
    if (pick.length > 1) multi.push(`${e.name}→${pick.join("/")}`);
    res.set(e.id, pick[0]);
  }
  return { res, unmatched, multi };
}

const mism = []; const add = (m, f, e, s) => mism.push({ id: m.id, sd: mm.res.get(m.id), f, e, s });
const fp = (m) => JSON.stringify(m);
const crit = (m) => (m.alwaysCrit ? "always" : m.highCritRatio ? "high" : "norm");
const sdCrit = (x) => (x.willCrit ? "always" : x.critRatio >= 2 ? "high" : "norm");
function sdChances(x) {
  const ss = x.secondaries || (x.secondary ? [x.secondary] : []);
  return ss.filter((s) => !(s.self && !s.status && !s.volatileStatus && !s.boosts)).map((s) => String(s.chance ?? 100)).sort().join(",");
}
function engChances(m) {
  const arr = [];
  for (const k of ["inflictsStatus", "inflictsVolatile"]) for (const s of m[k] || []) if (s.target !== "self") arr.push(String(s.chance ?? 100));
  for (const s of m.statChanges || []) if (s.target === "opponent") arr.push(String(s.chance ?? 100));
  return arr.sort().join(",");
}

  const sid = mm.res.get(m.id); const x = sdM[sid]; if (!x) continue;
  if (TYPE[m.type] !== x.type) add(m, "type", m.type, x.type);

  if (crit(m) !== sdCrit(x)) add(m, "crit", crit(m), `${sdCrit(x)}(critRatio ${x.critRatio}${x.willCrit ? ",willCrit" : ""})`);

  if (x.category !== "Status" && engChances(m) !== sdChances(x)) add(m, "secondary 확률", engChances(m) || "-", sdChances(x) || "-");
