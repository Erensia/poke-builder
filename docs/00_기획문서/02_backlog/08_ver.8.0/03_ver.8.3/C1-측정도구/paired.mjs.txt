// 8.3 C1 확정 분석(일회용): 판별 기록(c1-round.mjs 출력) 두 개를 (시드, 좌우)로 짝지어 후보 − 기준선의 짝 순이득, 갈라진 판 비율 f, 판 단위 부트스트랩 CI를 낸다.
// 사용: node paired.mjs <candidate-out.json> <baseline-out.json>   (각 파일 pairs[0].records 사용)
import { readFileSync } from "node:fs";
const [, , candFile, baseFile] = process.argv;
const cand = JSON.parse(readFileSync(candFile, "utf8")).pairs[0], base = JSON.parse(readFileSync(baseFile, "utf8")).pairs[0];
const key = (r) => `${r.s}|${r.side}`;
const bmap = new Map(base.records.map((r) => [key(r), r]));
const diffs = [], diverged = [];
let nc = 0, nb = 0, winC = 0, winB = 0, otherC = 0, otherB = 0, winsCD = 0, lossesCD = 0;
for (const r of cand.records) {
  const b = bmap.get(key(r));
  if (!b) continue;
  const d = r.win - b.win;
  diffs.push(d);
  if (r.h !== b.h) { diverged.push(d); if (d > 0) winsCD++; if (d < 0) lossesCD++; }
  winC += r.win === 1 ? 1 : 0; winB += b.win === 1 ? 1 : 0; otherC += r.win === 0.5 ? 1 : 0; otherB += b.win === 0.5 ? 1 : 0;
}
const n = diffs.length;
let seed = 99;
const rnd = () => { seed = (seed + 0x6d2b79f5) >>> 0; let t = seed; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
const mean = (a) => a.reduce((x, y) => x + y, 0) / a.length;
const rs = [];
for (let k = 0; k < 4000; k++) { let s = 0; for (let i = 0; i < n; i++) s += diffs[Math.floor(rnd() * n)]; rs.push(s / n); }
rs.sort((x, y) => x - y);
const rate = (w, o, total) => w / (total - o);
const pct = (x) => `${(100 * x).toFixed(2)}%p`;
console.log(`짝 ${n}판 | 후보 ${cand.label} ${(100 * rate(winC, otherC, n)).toFixed(1)}% · 기준선 ${base.label} ${(100 * rate(winB, otherB, n)).toFixed(1)}%`);
console.log(`갈라진 판 f = ${(100 * diverged.length / n).toFixed(1)}% (${diverged.length}/${n}) | 갈라진 판 안: 후보가 이김 ${winsCD} · 짐 ${lossesCD} · 평균 차 ${diverged.length ? pct(mean(diverged)) : "n/a"}`);
console.log(`짝 순이득(전체 판 대비) ${pct(mean(diffs))} 95% CI [${pct(rs[Math.floor(0.025 * rs.length)])}, ${pct(rs[Math.floor(0.975 * rs.length)])}]`);
const m = mean(diffs), lo = rs[Math.floor(0.025 * rs.length)];
console.log(`합격 규칙(설계서 §1): 짝 순이득 ≥ +1.5%p 그리고 CI 하한 > 0 → ${m >= 0.015 && lo > 0 ? "PASS" : "FAIL"} (후보 대 P6 ≥ 55%는 별도 확인)`);
