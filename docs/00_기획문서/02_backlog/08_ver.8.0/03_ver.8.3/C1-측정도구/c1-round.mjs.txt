// 8.3 C1 확정용 드라이버(일회용): zz-h2h-rec.mjs(판별 기록을 내는 h2h 사본)를 시드 조각으로 병렬 실행해 판별 기록을 합쳐 낸다.
// 사용: SIM_PATH=<zz-h2h-rec.mjs> node c1-round.mjs <pairs.json> <out.json> --from 1001 --seeds 1200 --chunk 20 --jobs 10
//   pairs.json = [{ label, a, b }]  승률은 A 기준(무승부 제외). --from = 시작 시드, --seeds = 시드 개수(1시드 = 2판)
import { spawn } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";
const args = process.argv.slice(2);
const opt = (name, fallback) => { const i = args.indexOf(`--${name}`); return i >= 0 ? args[i + 1] : fallback; };
const pairs = JSON.parse(readFileSync(args[0], "utf8"));
const outFile = args[1];
const from0 = Number(opt("from", 1)), seeds = Number(opt("seeds", 300)), chunk = Number(opt("chunk", 20)), jobs = Number(opt("jobs", 10));
const sim = process.env.SIM_PATH;
const queue = [];
for (let p = 0; p < pairs.length; p++) for (let off = 0; off < seeds; off += chunk) queue.push({ p, from: from0 + off, count: Math.min(chunk, seeds - off) });
const total = queue.length, records = pairs.map(() => []);
const started = Date.now();
let done = 0;
const run = ({ p, from, count }) => new Promise((resolve, reject) => {
  const env = { ...process.env, POOL: "samples", SELECT: "1", SEED_FROM: String(from) };
  const child = spawn(process.execPath, [sim, "h2h", String(count), pairs[p].a, pairs[p].b], { env });
  let out = "", err = "";
  child.stdout.on("data", (d) => (out += d));
  child.stderr.on("data", (d) => (err += d));
  child.on("close", (code) => {
    const line = out.trim().split("\n").filter((l) => l.startsWith("{")).pop();
    if (code !== 0 || !line) return reject(new Error(`${pairs[p].label} 시드 ${from}: exit ${code}\n${err.slice(0, 500)}`));
    records[p].push(...JSON.parse(line).records);
    // 조각마다 중간 저장 — 중단돼도 지금까지의 판별 기록이 남는다
    writeFileSync(outFile, JSON.stringify({ from: from0, seeds, done: done + 1, total, minutes: (Date.now() - started) / 60000, pairs: pairs.map((pr, i) => ({ label: pr.label, a: pr.a, b: pr.b, records: records[i] })) }));
    console.error(`[${++done}/${total}] ${pairs[p].label} 시드 ${from}~${from + count - 1} (${((Date.now() - started) / 60000).toFixed(1)}분)`);
    resolve();
  });
});
await Promise.all(Array.from({ length: Math.min(jobs, total) }, async () => { while (queue.length > 0) await run(queue.shift()); }));
console.log(`완료 ${done}/${total} · ${((Date.now() - started) / 60000).toFixed(1)}분`);
