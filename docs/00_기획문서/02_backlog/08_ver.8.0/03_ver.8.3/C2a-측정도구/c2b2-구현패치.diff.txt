+function mergeTemp(tb: OpponentTemp | undefined, tl: OpponentTemp | undefined, tr: OpponentTemp | undefined, dL: number, dR: number): OpponentTemp | undefined {
+  if (!tb && !tl && !tr) return undefined;
+  const vec = (x: OpponentTemp | undefined) => (x ? [...x.switch, ...x.move, x.nSwitch, x.nMove] : null);
+  const vb = vec(tb) ?? Array.from({ length: 2 * (tl ?? tr)!.switch.length + 2 }, () => 0);
